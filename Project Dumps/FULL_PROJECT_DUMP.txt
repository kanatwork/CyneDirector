# --- Crash Handler ---
def global_exception_handler(exctype, value, tb):
    """Captures crashes and saves them to a log file instead of just closing"""
    if exctype == KeyboardInterrupt:
        sys.__excepthook__(exctype, value, tb)
        return
    
    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    log_path = config.LOG_DIR / f"crash_{timestamp}.txt"
    
    error_msg = "".join(traceback.format_exception(exctype, value, tb))
    
    with open(log_path, "w") as f:
        f.write(f"CYNEDIRECTOR V20 CRASH REPORT\n{'='*30}\n")
        f.write(error_msg)
        
    print(f"CRITICAL ERROR: {value}")
    print(f"Report saved to: {log_path}")
    
    # Try to show a popup if QApplication is alive
    if QApplication.instance():
        QMessageBox.critical(None, "Critical Error", f"An error occurred.\nLog saved to: {log_path}\n\nError: {value}")

sys.excepthook = global_exception_handler

IGNORE_EXTS = {'.pyc', '.jpg', '.mp4', '.json', '.kan'}

def pack_project():
    output_file = "FULL_PROJECT_DUMP.txt"
    root_dir = os.getcwd()
    
    with open(output_file, "w", encoding="utf-8") as outfile:
        outfile.write("=== CYNEDIRECTOR CODEBASE DUMP ===\n\n")
        
        for dirpath, dirnames, filenames in os.walk(root_dir):
            # Skip ignored directories
            dirnames[:] = [d for d in dirnames if d not in IGNORE_DIRS]
            
            for filename in filenames:
                if any(filename.endswith(ext) for ext in IGNORE_EXTS):
                    continue
                    
                filepath = os.path.join(dirpath, filename)
                rel_path = os.path.relpath(filepath, root_dir)
                
                # Only grab .py files or requirements
                if filename.endswith(".py") or filename == "requirements.txt":
                    outfile.write(f"\n\n{'='*50}\nFILE: {rel_path}\n{'='*50}\n")
                    try:
                        with open(filepath, "r", encoding="utf-8") as infile:
                            outfile.write(infile.read())
                    except Exception as e:
                        outfile.write(f"[Error reading file: {e}]")

    print(f"Done! Upload '{output_file}' to the chat.")

if __name__ == "__main__":

import torch
import sys
import os

class AIBackend:
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super(AIBackend, cls).__new__(cls)
            
            # --- FORCE GPU CHECK ---
            if torch.cuda.is_available():
                cls._instance.device = "cuda"
                torch.backends.cudnn.benchmark = True 
                print(f"🚀 AI ACCELERATION: ON ({torch.cuda.get_device_name(0)})")
            else:
                cls._instance.device = "cpu"
                print("⚠️ WARNING: RUNNING ON CPU. INDEXING WILL BE SLOW.")

            cls._instance.clip_model = None
            cls._instance.clip_processor = None
            cls._instance.whisper_model = None
            cls._instance.tag_embeddings = None
            
        return cls._instance

    def load_clip(self):
        if self.clip_model: return self.clip_model, self.clip_processor
        
        print(f"Loading CLIP (Large) on {self.device}...")
        try:
            from transformers import CLIPProcessor, CLIPModel
            from core.tags import get_tag_bank
            
            # UPGRADE: Using ViT-Large-Patch14 (Much smarter than Base)
            model_name = "openai/clip-vit-large-patch14"
            
            self.clip_processor = CLIPProcessor.from_pretrained(model_name)
            self.clip_model = CLIPModel.from_pretrained(model_name).to(self.device)
            
            # Pre-compute Tag Embeddings
            print("Pre-computing Tag Embeddings...")
            tags = get_tag_bank()
            
            # Process tags in batches to avoid VRAM spikes
            tag_embeddings_list = []
            batch_size = 50
            for i in range(0, len(tags), batch_size):
                batch_tags = tags[i:i+batch_size]
                inputs = self.clip_processor(text=batch_tags, return_tensors="pt", padding=True).to(self.device)
                with torch.no_grad():
                    batch_embeds = self.clip_model.get_text_features(**inputs)
                    batch_embeds /= batch_embeds.norm(p=2, dim=-1, keepdim=True)
                    tag_embeddings_list.append(batch_embeds)
            
            self.tag_embeddings = torch.cat(tag_embeddings_list)
                
            return self.clip_model, self.clip_processor

        except Exception as e:
            print(f"AI LOAD ERROR: {e}")
            raise e

    def load_whisper(self):
        if self.whisper_model: return self.whisper_model
        
        print(f"Loading Faster-Whisper (Large-v3) on {self.device}...")
        from faster_whisper import WhisperModel
        
        # UPGRADE: Using 'large-v3' for maximum accuracy.
        # compute_type="float16" enables GPU acceleration.
        try:
            device_str = "cuda" if self.device == "cuda" else "cpu"
            compute_type = "float16" if self.device == "cuda" else "int8"
            
            self.whisper_model = WhisperModel("large-v3", device=device_str, compute_type=compute_type)
        except Exception as e:
            print(f"Whisper Load Error: {e}. Falling back to medium model.")
            self.whisper_model = WhisperModel("medium", device=device_str, compute_type="int8")
            
        return self.whisper_model

import chromadb
import json
import os
import uuid
from pathlib import Path

class Database:
    _instance = None
    
    def __new__(cls, project_path=None):
        if cls._instance is None:
            cls._instance = super(Database, cls).__new__(cls)
            cls._instance.client = None
            cls._instance.project_path = project_path
        return cls._instance

    def initialize(self, project_path):
        self.project_path = project_path
        db_path = os.path.join(project_path, "_cyne_db")
        os.makedirs(db_path, exist_ok=True)
        
        # Initialize Vector DB (Chroma)
        self.client = chromadb.PersistentClient(path=db_path)
        
        # Collections
        # We use cosine distance for search relevance
        self.visuals = self.client.get_or_create_collection(name="visual_embeddings", metadata={"hnsw:space": "cosine"})
        self.faces = self.client.get_or_create_collection(name="face_embeddings", metadata={"hnsw:space": "cosine"})
        self.transcripts = self.client.get_or_create_collection(name="transcripts", metadata={"hnsw:space": "cosine"})

    def add_visual_embeddings(self, video_path, vectors, timestamps):
        """
        Saves a batch of AI visual vectors to the database.
        """
        if not vectors: return
        
        # Generate unique IDs for every frame: "C:/Videos/Clip1.mp4_12.5"
        ids = [f"{video_path}_{t}" for t in timestamps]
        
        # Metadata allows us to filter by file later
        metadatas = [{"source": video_path, "timestamp": t} for t in timestamps]
        
        try:
            self.visuals.upsert(
                ids=ids,
                embeddings=vectors,
                metadatas=metadatas
            )
            # print(f"DEBUG: Saved {len(vectors)} vectors for {os.path.basename(video_path)}")
        except Exception as e:
            print(f"DB ERROR: {e}")

    def save_video_tags(self, video_path, tags, summary_or_text):
        """
        Saves metadata to a JSON sidecar file.
        """
        meta_path = f"{video_path}.json"
        
        # 1. Load existing
        data = {}
        if os.path.exists(meta_path):
            try:
                with open(meta_path, 'r') as f:
                    data = json.load(f)
            except: pass

        # 2. Update
        if tags:
            data["tags"] = tags
            data["summary"] = summary_or_text
        else:
            data["transcript"] = summary_or_text
            
        data["last_scanned"] = str(os.path.getmtime(video_path))
        
        # 3. Save
        with open(meta_path, 'w') as f:
            json.dump(data, f, indent=4)

    def update_metadata_key(self, video_path, key, value):
        """
        Surgically updates a specific key in the JSON sidecar.
        Useful for adding 'faces' without overwriting 'transcript'.
        """
        meta_path = f"{video_path}.json"
        data = {}
        
        # 1. Load existing
        if os.path.exists(meta_path):
            try:
                with open(meta_path, 'r') as f:
                    data = json.load(f)
            except: pass
            
        # 2. Update specific field
        data[key] = value
        data["last_scanned"] = str(os.path.getmtime(video_path))
        
        # 3. Save
        with open(meta_path, 'w') as f:
            json.dump(data, f, indent=4)

import os
import pickle
import json
import numpy as np
from pathlib import Path

class FaceDB:
    def __init__(self, project_path):
        self.project_path = project_path
        self.db_dir = os.path.join(project_path, "_cyne_db", "faces")
        self.encodings_path = os.path.join(self.db_dir, "encodings.pkl")
        self.names_path = os.path.join(self.db_dir, "names.json")
        
        # Ensure directory exists
        os.makedirs(self.db_dir, exist_ok=True)
        
        # In-Memory Cache
        self.known_encodings = []
        self.known_ids = []
        self.id_to_name = {}
        
        self.load()

    def load(self):
        """Loads encodings and names from disk."""
        # 1. Load Names Map
        if os.path.exists(self.names_path):
            try:
                with open(self.names_path, 'r') as f:
                    self.id_to_name = json.load(f)
            except:
                self.id_to_name = {}

        # 2. Load Encodings (The "Brain")
        if os.path.exists(self.encodings_path):
            try:
                with open(self.encodings_path, 'rb') as f:
                    data = pickle.load(f)
                    self.known_ids = data.get("ids", [])
                    self.known_encodings = data.get("encodings", [])
            except:
                self.known_ids = []
                self.known_encodings = []

    def save_encodings(self):
        """Saves the AI data to disk."""
        data = {
            "ids": self.known_ids,
            "encodings": self.known_encodings
        }
        with open(self.encodings_path, 'wb') as f:
            pickle.dump(data, f)

    def save_names(self):
        """Saves the Human Readable names to disk."""
        with open(self.names_path, 'w') as f:
            json.dump(self.id_to_name, f, indent=4)

    def add_face(self, person_id, encoding):
        """Registers a new face."""
        self.known_ids.append(person_id)
        self.known_encodings.append(encoding)
        self.id_to_name[person_id] = person_id # Default name is ID
        
        self.save_encodings()
        self.save_names()

    def rename_person(self, person_id, new_name):
        """Updates the display name for a person."""
        if person_id in self.id_to_name:
            self.id_to_name[person_id] = new_name
            self.save_names()

    def get_name(self, person_id):
        return self.id_to_name.get(person_id, person_id)

    def get_next_id(self):
        return f"Person_{len(self.known_ids) + 1}"

import os
import json
import torch
from core.database import Database
from core.ai_models import AIBackend
from core.face_db import FaceDB  # <--- NEW IMPORT

class SearchEngine:
    def __init__(self, project_path):
        self.project_path = project_path
        self.db = Database(project_path)
        self.ai = AIBackend()
        self.face_db = FaceDB(project_path) # <--- NEW
        
        # RAM CACHE (The Speed Boost)
        self.cache = {} 
        self.build_index()

    def build_index(self):
        """
        Scans the ENTIRE project and loads metadata into RAM.
        """
        print("Building Search Index...")
        for root, dirs, files in os.walk(self.project_path):
            if "_cyne_db" in root: continue 
            
            for file in files:
                if file.endswith(".json"):
                    meta_path = os.path.join(root, file)
                    video_path = meta_path.replace(".json", "") 
                    
                    if not os.path.exists(video_path): continue
                    
                    try:
                        with open(meta_path, 'r') as f:
                            data = json.load(f)
                            self.cache[video_path] = data
                    except:
                        continue
        print(f"Index built. Loaded {len(self.cache)} items.")

    def resolve_name_to_ids(self, query_name):
        """
        Returns a list of IDs (e.g. ['Person_1']) that match the query name.
        """
        query_name = query_name.lower()
        matched_ids = []
        
        # FaceDB.id_to_name is { "Person_1": "John Doe", "Person_2": "Alice" }
        for person_id, name in self.face_db.id_to_name.items():
            if query_name in name.lower():
                matched_ids.append(person_id)
                
        return matched_ids

    def search(self, query):
        """
        Instant search using RAM cache + Vector DB + Rich Metadata + FACES
        """
        query_terms = query.lower().strip().split()
        full_query = query.lower().strip()
        if not query_terms: return []

        results = []
        seen_paths = set()
        
        # --- PRE-CALC: NAME RESOLUTION ---
        target_person_ids = self.resolve_name_to_ids(full_query)
        # ---------------------------------

        # --- 1. VECTOR SEARCH (AI Meaning) ---
        if self.ai.clip_model:
            try:
                inputs = self.ai.clip_processor(text=[query], return_tensors="pt", padding=True).to(self.ai.device)
                with torch.no_grad():
                    text_features = self.ai.clip_model.get_text_features(**inputs)
                    text_features /= text_features.norm(p=2, dim=-1, keepdim=True)
                
                query_vector = text_features.cpu().numpy()[0].tolist()

                vec_res = self.db.visuals.query(
                    query_embeddings=[query_vector],
                    n_results=20,
                    include=["metadatas", "distances"]
                )
                
                if vec_res['ids'] and vec_res['ids'][0]:
                    for i, _id in enumerate(vec_res['ids'][0]):
                        metadata = vec_res['metadatas'][0][i]
                        distance = vec_res['distances'][0][i]
                        score = max(0, (1 - distance) * 100)
                        
                        if score < 15: continue
                        
                        path = metadata['source']
                        timestamp = metadata.get('timestamp', 0)
                        
                        results.append({
                            "path": path,
                            "match_type": "VISUAL (AI)",
                            "context": f"Visual match at {int(timestamp)}s",
                            "score": score,
                            "timestamp": timestamp
                        })
                        seen_paths.add(path)
            except Exception as e:
                print(f"Vector search skip: {e}")

        # --- 2. TEXT & METADATA SEARCH (RAM Cache) ---
        for video_path, data in self.cache.items():
            score = 0
            match_type = ""
            context = ""
            timestamp = 0
            
            filename = os.path.basename(video_path).lower()

            # A. Filename Match
            if full_query in filename:
                score += 100
                match_type = "FILENAME"
                context = "Filename match"

            # B. FACE SEARCH (Highest Priority)
            elif target_person_ids and "faces" in data:
                video_faces = data["faces"]
                for target_id in target_person_ids:
                    if target_id in video_faces:
                        score = 200 # Boost above almost everything else
                        match_type = "CAST"
                        human_name = self.face_db.get_name(target_id)
                        context = f"Featuring: {human_name}"
                        break

            # C. Transcript
            elif "transcript" in data:
                trans_data = data["transcript"]
                if isinstance(trans_data, list):
                    for seg in trans_data:
                        if full_query in seg["text"].lower():
                            score = 80
                            match_type = "DIALOGUE"
                            timestamp = seg["start"]
                            context = f"Says: \"{seg['text']}\""
                            break 
                elif isinstance(trans_data, str) and full_query in trans_data.lower():
                    score = 60
                    match_type = "DIALOGUE"
                    context = "Keyword found in audio"

            # D. Visual Tags
            elif "tags" in data:
                tags = [t.lower() for t in data["tags"]]
                for tag in tags:
                    if full_query in tag:
                        score += 50
                        match_type = "TAG"
                        context = f"Tagged: {tag}"
                        break
            
            if score > 0 and video_path not in seen_paths:
                results.append({
                    "path": video_path,
                    "match_type": match_type,
                    "context": context,
                    "score": score,
                    "timestamp": timestamp
                })

        # Sort by Score
        results.sort(key=lambda x: x['score'], reverse=True)
        return results

class FaceCard(QFrame):
    # Signal: Card ID (Person ID) requested deletion
    delete_requested = pyqtSignal(str) 
    # Signal: Card ID requested a rename -> (id, new_name)
    rename_requested = pyqtSignal(str, str)

    def __init__(self, person_id, image, name="Unknown"):
        super().__init__()
        self.person_id = person_id
        self.current_name = name
        
        self.setFixedSize(130, 160)
        self.setStyleSheet("""
            FaceCard { background: #252526; border-radius: 8px; border: 1px solid #333; }
            FaceCard:hover { border: 1px solid #666; background: #2A2A2A; }
        """)
        
        layout = QVBoxLayout(self)
        layout.setContentsMargins(10, 10, 10, 5)
        
        # Image
        self.img_lbl = QLabel()
        # Handle both QImage and QPixmap
        if isinstance(image, QImage):
            pix = QPixmap.fromImage(image)
        else:
            pix = image
            
        pix = pix.scaled(90, 90, Qt.AspectRatioMode.KeepAspectRatioByExpanding, Qt.TransformationMode.SmoothTransformation)
        self.img_lbl.setPixmap(pix)
        self.img_lbl.setFixedSize(90, 90)
        self.img_lbl.setStyleSheet("border-radius: 45px; background: black; border: 2px solid #333;")
        layout.addWidget(self.img_lbl, alignment=Qt.AlignmentFlag.AlignCenter)
        
        # Name
        self.name_lbl = QLabel(name)
        self.name_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.name_lbl.setStyleSheet("color: white; font-weight: bold; font-size: 11px; margin-top: 5px; border: none;")
        layout.addWidget(self.name_lbl)

    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            # Rename on Left Click
            new_name, ok = QInputDialog.getText(self, "Rename", f"Rename {self.current_name} to:")
            if ok and new_name:
                self.name_lbl.setText(new_name)
                self.current_name = new_name
                # Signal the change to the parent
                self.rename_requested.emit(self.person_id, new_name)

    def contextMenuEvent(self, event):
        # Right Click Menu
        menu = QMenu(self)
        menu.setStyleSheet("QMenu { background: #333; color: white; border: 1px solid #555; } QMenu::item:selected { background: #555; }")
        
        delete_action = QAction("Delete Person", self)
        delete_action.triggered.connect(self.request_delete)
        menu.addAction(delete_action)
        
        menu.exec(event.globalPos())

    def request_delete(self):
        # Ask for confirmation
        confirm = QMessageBox.question(
            self, "Confirm Delete", 
            f"Are you sure you want to delete '{self.current_name}'?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )
        if confirm == QMessageBox.StandardButton.Yes:
            self.delete_requested.emit(self.person_id)

class FacesTab(QWidget):
    def __init__(self):
        super().__init__()
        layout = QVBoxLayout(self)
        
        header = QLabel("DETECTED PEOPLE")
        header.setStyleSheet("color: #888; font-weight: bold; margin-bottom: 10px;")
        layout.addWidget(header)
        
        scroll = QScrollArea()
        scroll.setWidgetResizable(True)
        scroll.setStyleSheet("background: #121212; border: none;")
        
        self.grid_container = QWidget()
        self.grid_layout = QGridLayout(self.grid_container)
        self.grid_layout.setAlignment(Qt.AlignmentFlag.AlignTop | Qt.AlignmentFlag.AlignLeft)
        self.grid_layout.setSpacing(15)
        
        scroll.setWidget(self.grid_container)
        layout.addWidget(scroll)
        
        self.faces_count = 0
        self.cards = {} # Keep track of widgets: {person_id: card_widget}
        
        # Persistence
        self.project_path = None
        self.face_db = None

    def set_project_path(self, path):
        """Called by Main Window on startup to initialize DB connection."""
        self.project_path = path
        self.face_db = FaceDB(path)
        self.load_existing_faces()

    def load_existing_faces(self):
        """Populates grid from disk on startup."""
        if not self.face_db: return
        
        # Iterate through known IDs in the DB
        for pid in self.face_db.known_ids:
            if pid in self.cards: continue
            
            name = self.face_db.get_name(pid)
            thumb_path = os.path.join(self.face_db.db_dir, f"{pid}.jpg")
            
            if os.path.exists(thumb_path):
                image = QImage(thumb_path)
                if not image.isNull():
                    self.add_face_card(pid, name, image)

    def add_face_card(self, person_id, name, q_image):
        if person_id in self.cards: return

        # Calculate grid position (5 columns wide)
        row = self.faces_count // 5
        col = self.faces_count % 5
        
        card = FaceCard(person_id, q_image, name)
        
        # Connect signals
        card.delete_requested.connect(self.delete_face)
        card.rename_requested.connect(self.handle_rename)
        
        self.grid_layout.addWidget(card, row, col)
        self.cards[person_id] = card
        
        self.faces_count += 1

    def handle_rename(self, person_id, new_name):
        """Saves the new name to the database."""
        if self.face_db:
            self.face_db.rename_person(person_id, new_name)
            # Optional: Feedback log
            print(f"DEBUG: Renamed {person_id} to {new_name}")

    def delete_face(self, person_id):
        if person_id in self.cards:
            widget = self.cards[person_id]
            
            # 1. Remove from Layout
            self.grid_layout.removeWidget(widget)
            widget.deleteLater()
            del self.cards[person_id]
            
            # 2. Delete from Disk and DB
            if self.face_db:
                # Ideally, FaceDB should handle file deletion too, but we do it here for now

    # --- TRANSCRIPT LOGIC ---
    def update_preview_panel(self):
        paths = self.tree.get_selected_file_paths()
        if not paths: 
            self.preview_lbl.clear()
            self.meta_box.clear()
            self.current_preview_path = None
            return
            
        file_path = paths[0]
        self.current_preview_path = file_path # Store for player
        
        # 1. Load Thumbnail
        try:
            cap = cv2.VideoCapture(file_path)
            total = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
            if total > 100: cap.set(cv2.CAP_PROP_POS_FRAMES, 100)
            ret, frame = cap.read()
            cap.release()
            
            if ret:
                frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                h, w, ch = frame.shape
                bytes_per_line = ch * w
                qt_img = QImage(frame.data, w, h, bytes_per_line, QImage.Format.Format_RGB888)
                pix = QPixmap.fromImage(qt_img)
                scaled_pix = pix.scaled(self.preview_lbl.size(), Qt.AspectRatioMode.KeepAspectRatio, Qt.TransformationMode.SmoothTransformation)
                self.preview_lbl.setPixmap(scaled_pix)
        except:
            self.preview_lbl.setText("Preview unavailable")

        # 2. Load Rich Metadata
        json_path = f"{file_path}.json"
        html = f"<h3 style='color:{COLORS['accent']}'>{os.path.basename(file_path)}</h3>"
        
        if os.path.exists(json_path):
            try:
                with open(json_path, 'r') as f:
                    data = json.load(f)
                    
                    # TAGS
                    tags = data.get("tags", [])
                    if tags:
                        html += f"<b>[ VISUALS ]</b><br><span style='color:#AAA'>{', '.join(tags)}</span><br><br>"
                    
                    # TRANSCRIPT
                    transcript_data = data.get("transcript", "")
                    
                    if isinstance(transcript_data, list):
                        # RICH METADATA FOUND!
                        html += "<b>[ AUDIO TRANSCRIPT (Click to Play) ]</b><br>"
                        for seg in transcript_data:
                            start = seg['start']
                            text = seg['text']
                            # Create a link with the timestamp as the href
                            html += f"<a href='{start}' style='color:{COLORS['accent']}; text-decoration:none;'><b>[{self.fmt_time(start)}]</b></a> {text} "
                    
                    elif isinstance(transcript_data, str) and transcript_data:
                        # Legacy String
                        html += f"<b>[ TRANSCRIPT ]</b><br>{transcript_data}"
                        
                    summary = data.get("summary", "")
                    if summary and not transcript_data:
                         html += f"<b>[ SUMMARY ]</b><br>{summary}"

            except Exception as e:
                html += f"<br><i style='color:red'>Error: {e}</i>"
        else:
            html += "<i>Not indexed yet.</i>"
            
        self.meta_box.setHtml(html)

    def fmt_time(self, seconds):
        """Converts 65.5 -> 01:05"""
        m, s = divmod(seconds, 60)
        return f"{int(m):02}:{int(s):02}"

    def handle_transcript_click(self, url):

        try:
            timestamp = float(url.toString())
            if self.current_preview_path:

                    self.player_window = PlayerWindow()
                
                self.player_window.load_video(self.current_preview_path, timestamp)

            print(f"Jump Error: {e}")

from PyQt6.QtWidgets import QTreeWidget, QTreeWidgetItem, QAbstractItemView, QHeaderView

class MediaTree(QTreeWidget):
    def __init__(self):
        super().__init__()
        self.setHeaderLabels(["FILENAME", "RESOLUTION", "DURATION", "FPS", "FACES", "AI SUMMARY", "FULL_PATH"])
        
        header = self.header()

        header.setSectionResizeMode(0, QHeaderView.ResizeMode.ResizeToContents)
        self.setColumnHidden(6, True) 
        self.setAlternatingRowColors(True)
        self.setSelectionMode(QAbstractItemView.SelectionMode.ExtendedSelection)
        self.setDragEnabled(True) 
        
        # Connect Clicks
        self.itemClicked.connect(self.handle_click)
        self.is_updating = False

        self.setStyleSheet(f"""
            QTreeWidget {{ background: {COLORS['bg_panel']}; border: 1px solid #333; border-radius: 4px; font-size: 13px; color: #DDD; }}
            QHeaderView::section {{ background: #252526; padding: 6px; border: none; border-bottom: 1px solid #444; color: #BBB; font-weight: bold; }}
            QTreeWidget::item {{ padding: 6px; }}
            QTreeWidget::item:selected {{ background: #333; color: {COLORS['accent']}; border-left: 2px solid {COLORS['accent']}; }}
        """)

    def handle_click(self, item, column):
        """Recursively toggles children if a folder checkbox is clicked"""
        if column == 0 and not self.is_updating:
            self.is_updating = True
            new_state = item.checkState(0)
            self._set_children_state(item, new_state)
            self.is_updating = False

    def mouseReleaseEvent(self, event):
        """
        Custom handler to support bulk checking.
        If user clicks a checkbox while multiple rows are selected, toggle all of them.
        """
        # 1. Let the default handler process the click (toggles the specific item clicked)
        super().mouseReleaseEvent(event)
        
        # 2. Find out where we clicked
        item = self.itemAt(event.pos())
        if not item: return
        
        # Check if click was roughly in the checkbox area (Column 0, left side)
        # Visual rect of the item
        rect = self.visualItemRect(item)
        # Checkbox is usually in the first 40 pixels of the item rect
        click_x_relative = event.pos().x() - rect.x()
        
        # If clicked on the checkbox (heuristic: x < 40) AND we have a multi-selection
        if click_x_relative < 40 and len(self.selectedItems()) > 1:
            
            # Get the new state of the item we just clicked
            new_state = item.checkState(0)
            
            # Apply to all other selected items
            self.is_updating = True # Prevent recursion triggers if you have them
            for sel_item in self.selectedItems():
                if sel_item is not item: # Skip the one we already clicked
                    sel_item.setCheckState(0, new_state)
            self.is_updating = False

    def _set_children_state(self, parent_item, state):
        """Helper to recursively check/uncheck children"""
        for i in range(parent_item.childCount()):
            child = parent_item.child(i)
            child.setCheckState(0, state)
            self._set_children_state(child, state)

    def toggle_all(self, state=True):
        self.is_updating = True
        root = self.invisibleRootItem()
        target = Qt.CheckState.Checked if state else Qt.CheckState.Unchecked
        for i in range(root.childCount()):
            item = root.child(i)
            item.setCheckState(0, target)
            self._set_children_state(item, target)
        self.is_updating = False

    # --- FOLDER IMPORT LOGIC ---
    def add_folder(self, folder_path):
        """Starts the recursive add from the root level"""
        if not os.path.exists(folder_path): return
        
        folder_name = os.path.basename(folder_path)
        root_item = QTreeWidgetItem(self)
        root_item.setText(0, f"📁 {folder_name}")
        root_item.setText(6, folder_path)
        root_item.setFlags(root_item.flags() | Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsSelectable)
        root_item.setCheckState(0, Qt.CheckState.Checked)
        
        # Recursive Dive
        self._add_items_recursive(folder_path, root_item)
        root_item.setExpanded(True)

    def _add_items_recursive(self, parent_path, parent_item):
        try:
            for entry in os.scandir(parent_path):
                if entry.is_dir():
                    # Create Folder Node
                    folder_item = QTreeWidgetItem(parent_item)
                    folder_item.setText(0, f"📁 {entry.name}")
                    folder_item.setText(6, entry.path)
                    folder_item.setFlags(folder_item.flags() | Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsSelectable)
                    folder_item.setCheckState(0, Qt.CheckState.Checked)
                    
                    # Recurse
                    self._add_items_recursive(entry.path, folder_item)
            
                elif entry.is_file() and entry.name.lower().endswith(('.mp4', '.mov', '.mxf', '.braw', '.avi')):
                    # Create File Node
                    self._create_file_item(entry.path, parent_item)
        except PermissionError:
            pass

    def add_files_flat(self, file_paths):
        """Legacy flat import for individual file selection"""
        existing_paths = set(self.get_all_file_paths())
        for path in file_paths:
            if path in existing_paths: continue
            self._create_file_item(path, self)

    def _create_file_item(self, path, parent):
        name = os.path.basename(path)
        res_str, dur_str, fps_str = "Unknown", "--:--", "--"
        summary_str = "Waiting for Index..."
        
        # Read Meta
        try:
            cap = cv2.VideoCapture(path)
            if cap.isOpened():
                w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
                h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
                fps = cap.get(cv2.CAP_PROP_FPS)
                frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
                res_str = f"{w}x{h}"
                fps_str = f"{fps:.2f}"
                if fps > 0: dur_str = str(datetime.timedelta(seconds=int(frames/fps)))
            cap.release()
        except: pass
        
        # Read AI Sidecar
        json_path = f"{path}.json"
        if os.path.exists(json_path):
            try:
                with open(json_path, 'r') as f:
                    data = json.load(f)
                    if "summary" in data and data["summary"]:
                        summary_str = data["summary"]
                    elif "transcript" in data:
                        t = data["transcript"]
                        if isinstance(t, list): summary_str = "Transcript Available (Rich)"
                        else: summary_str = t[:50] + "..."
            except: pass  # <--- FIXED INDENTATION

        item = QTreeWidgetItem(parent)
        item.setFlags(item.flags() | Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsSelectable | Qt.ItemFlag.ItemIsEnabled | Qt.ItemFlag.ItemIsDragEnabled)
        item.setCheckState(0, Qt.CheckState.Checked)
        item.setText(0, name)
        item.setText(1, res_str)
        item.setText(2, dur_str)
        item.setText(3, fps_str)
        item.setText(4, "0")
        item.setText(5, summary_str)
        item.setText(6, path)

    # --- RETRIEVAL HELPERS ---
    def get_checked_file_paths(self):
        paths = []
        root = self.invisibleRootItem()
        self._collect_checked_recursive(root, paths)
        return paths

    def _collect_checked_recursive(self, parent, paths):
        for i in range(parent.childCount()):
            item = parent.child(i)
            # If it's a file (has metadata in col 1) and checked
            if item.checkState(0) == Qt.CheckState.Checked and item.childCount() == 0 and item.text(1) != "":
                paths.append(item.text(6))
            # Recurse
            self._collect_checked_recursive(item, paths)

    def get_all_file_paths(self):
        # Used for Saving Project
        paths = []
        root = self.invisibleRootItem()
        all_p = []
        def _recurse(p_item):
            for i in range(p_item.childCount()):
                it = p_item.child(i)
                if it.childCount() == 0 and it.text(1) != "": all_p.append(it.text(6))
                _recurse(it)
        _recurse(root)
        return all_p

    def get_selected_file_paths(self):
        # Used for Preview
        return [item.text(6) for item in self.selectedItems() if item.childCount() == 0]

    def update_item_status(self, file_path, column_index, new_text):
        # Recursive search for item
        root = self.invisibleRootItem()
        def _find_and_update(parent):
            for i in range(parent.childCount()):
                it = parent.child(i)
                if it.text(6) == file_path:
                    it.setText(column_index, str(new_text))
                    return True
                if _find_and_update(it): return True
            return False
        _find_and_update(root)

import os
import cv2
import face_recognition
import numpy as np
from PyQt6.QtCore import QThread, pyqtSignal
from PyQt6.QtGui import QImage
from core.face_db import FaceDB
from core.database import Database  # <--- NEW IMPORT

def sanitize_image(image):
    """Ensures image is contiguous and RGB (safe for face_recognition)."""
    if image is None: return None
    if not image.flags['C_CONTIGUOUS']:
        image = np.ascontiguousarray(image)
    return cv2.cvtColor(image, cv2.COLOR_BGR2RGB)

class FaceScannerWorker(QThread):
    log_signal = pyqtSignal(str)
    progress_signal = pyqtSignal(int)
    new_face_signal = pyqtSignal(str, str, object) 
    face_count_signal = pyqtSignal(str, int)       
    finished_signal = pyqtSignal()

    def __init__(self, file_paths, project_path):
        super().__init__()
        self.file_paths = file_paths
        self.project_path = project_path
        self.is_running = True
        
        # Initialize Persistent DB
        self.face_db = FaceDB(project_path)
        self.db = Database(project_path) # <--- NEW: Generic DB connection
        
        # Load known faces into local memory for fast matching
        self.known_encodings = self.face_db.known_encodings
        self.known_ids = self.face_db.known_ids

    def run(self):
        self.log_signal.emit(f"Loaded {len(self.known_ids)} known faces from database.")
        
        file_steps = []
        total_steps_global = 0
        
        # 1. Pre-calculate total work (frames to scan)
        for video_path in self.file_paths:
            if not self.is_running: break
            cap = cv2.VideoCapture(video_path)
            if cap.isOpened():
                fps = cap.get(cv2.CAP_PROP_FPS)
                frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
                # Scan roughly every 0.5 seconds
                step = int(fps * 0.5) if fps > 0 else 30
                if frames > 0:
                    steps = frames // step
                    file_steps.append((video_path, step, frames))
                    total_steps_global += steps
            cap.release()
            
        if total_steps_global == 0:
            self.finished_signal.emit()
            return

        current_step_global = 0
        faces_dir = self.face_db.db_dir 

        # 2. Main Scanning Loop
        for (video_path, step, total_frames) in file_steps:
            if not self.is_running: break
            
            self.log_signal.emit(f"Scanning: {os.path.basename(video_path)}")
            cap = cv2.VideoCapture(video_path)
            unique_faces_in_file = set() 
            
            for i in range(0, total_frames, step):
                if not self.is_running: break
                
                cap.set(cv2.CAP_PROP_POS_FRAMES, i)
                ret, frame = cap.read()
                if not ret or frame is None: continue
                
                try:
                    # Optimize Size
                    h, w = frame.shape[:2]
                    if w > 1080:
                        scale = 1080 / w
                        frame = cv2.resize(frame, (0,0), fx=scale, fy=scale)
                    
                    clean_image = sanitize_image(frame)

                    # Detect
                    face_locations = face_recognition.face_locations(clean_image, model="hog")
                    
                    if face_locations:
                        face_encodings = face_recognition.face_encodings(clean_image, face_locations)
                        
                        for (top, right, bottom, left), encoding in zip(face_locations, face_encodings):
                            # Compare against known faces
                            matches = face_recognition.compare_faces(self.known_encodings, encoding, tolerance=0.6)
                            person_id = None
                            
                            if True in matches:
                                first_match_index = matches.index(True)
                                person_id = self.known_ids[first_match_index]
                            else:
                                # NEW FACE FOUND
                                person_id = self.face_db.get_next_id()
                                
                                # Update Database
                                self.face_db.add_face(person_id, encoding)
                                
                                # Update Local Lists
                                self.known_encodings = self.face_db.known_encodings
                                self.known_ids = self.face_db.known_ids
                                
                                # Thumbnail Generation
                                face_slice = clean_image[top:bottom, left:right]
                                face_slice = np.ascontiguousarray(face_slice)
                                
                                # Save Thumbnail to Disk
                                thumb_path = os.path.join(faces_dir, f"{person_id}.jpg")
                                cv2.imwrite(thumb_path, cv2.cvtColor(face_slice, cv2.COLOR_RGB2BGR))
                                
                                # Send to UI
                                h_f, w_f, ch_f = face_slice.shape
                                bytes_per_line = ch_f * w_f
                                q_img = QImage(face_slice.data.tobytes(), w_f, h_f, bytes_per_line, QImage.Format.Format_RGB888).copy()
                                
                                self.new_face_signal.emit(person_id, person_id, q_img)
                            
                            unique_faces_in_file.add(person_id)

                except Exception as e:
                    pass 

                current_step_global += 1
                percent = int((current_step_global / total_steps_global) * 100)
                self.progress_signal.emit(percent)
            
            cap.release()
            
            # --- NEW: Save Detected Faces to Video Metadata ---
            if unique_faces_in_file:
                found_ids = list(unique_faces_in_file)
                self.db.update_metadata_key(video_path, "faces", found_ids)
                self.log_signal.emit(f"   Saved {len(found_ids)} faces to metadata.")
            # --------------------------------------------------

            self.face_count_signal.emit(video_path, len(unique_faces_in_file))
            
        self.finished_signal.emit()

    def stop(self):
        self.is_running = False

import os
from PyQt6.QtCore import QThread, pyqtSignal

class FolderImportWorker(QThread):
    progress_signal = pyqtSignal(str) # Emits current folder name being scanned
    finished_signal = pyqtSignal(list) # Emits the full list of found video files

    def __init__(self, folder_path):
        super().__init__()
        self.folder_path = folder_path
        # Extensions to look for
        self.valid_exts = {'.mp4', '.mov', '.mxf', '.braw', '.avi'}

    def run(self):
        found_files = []
        
        # Walk the directory tree
        for root, dirs, files in os.walk(self.folder_path):
            # Emit progress (just the folder name to keep UI clean)
            self.progress_signal.emit(f"Scanning: {os.path.basename(root)}")
            
            for file in files:
                if os.path.splitext(file)[1].lower() in self.valid_exts:
                    full_path = os.path.join(root, file)
                    found_files.append(full_path)
        
        self.finished_signal.emit(found_files)

from PyQt6.QtCore import QThread, pyqtSignal
import cv2
import torch
from PIL import Image
from core.ai_models import AIBackend
from core.database import Database
from core.tags import get_tag_bank
from collections import Counter

class IndexerWorker(QThread):
    log_signal = pyqtSignal(str)           
    progress_signal = pyqtSignal(int)      
    finished_signal = pyqtSignal()
    summary_signal = pyqtSignal(str, str)  
    
    def __init__(self, file_paths, project_path):
        super().__init__()
        self.file_paths = file_paths
        self.project_path = project_path
        self.is_running = True
        # BATCH SIZE: 32 is a sweet spot for RTX 5070 with ViT-Large
        self.batch_size = 32 

    def run(self):
        self.log_signal.emit("Initializing AI Core (High Accuracy Mode)...")
        ai = AIBackend()
        db = Database(self.project_path)
        
        try:
            model, processor = ai.load_clip()
        except Exception as e:
            self.log_signal.emit(f"AI Load Error: {e}")
            self.finished_signal.emit()
            return
            
        tag_list = get_tag_bank()
        total_files = len(self.file_paths)
        
        for idx, video_path in enumerate(self.file_paths):
            if not self.is_running: break
            
            self.log_signal.emit(f"Indexing: {video_path}")
            
            cap = cv2.VideoCapture(video_path)
            if not cap.isOpened(): continue
            
            fps = cap.get(cv2.CAP_PROP_FPS)
            total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
            
            # Scan every 1 second (High Density) or 2 seconds (Speed)
            step = int(fps * 1.5) if fps > 0 else 30
            
            frames_batch = []
            timestamps_batch = []
            all_detected_indices = []
            
            # Iterate through video
            for frame_num in range(0, total_frames, step):
                if not self.is_running: break
                
                cap.set(cv2.CAP_PROP_POS_FRAMES, frame_num)
                ret, frame = cap.read()
                if not ret: break
                
                # Pre-process image
                img = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                pil_img = Image.fromarray(img)
                
                frames_batch.append(pil_img)
                timestamps_batch.append(frame_num / fps if fps > 0 else 0)
                
                # --- PROCESS BATCH WHEN FULL ---
                if len(frames_batch) >= self.batch_size:
                    self.process_batch(frames_batch, timestamps_batch, ai, db, video_path, model, processor, all_detected_indices)
                    frames_batch = []
                    timestamps_batch = []
                
                # UI Progress
                current_percent = int(((idx) / total_files * 100) + (frame_num / total_frames * (100 / total_files)))
                self.progress_signal.emit(current_percent)
            
            # Process remaining frames
            if frames_batch:
                self.process_batch(frames_batch, timestamps_batch, ai, db, video_path, model, processor, all_detected_indices)
            
            cap.release()
            
            # --- Generate Summary ---
            if all_detected_indices:
                counts = Counter(all_detected_indices)
                top_common = counts.most_common(5) 
                summary_tags = [tag_list[i] for i, count in top_common]
                summary_text = f"Contains: {', '.join(summary_tags)}"
            else:
                summary_tags = []
                summary_text = "No clear subjects."
            
            db.save_video_tags(video_path, summary_tags, summary_text)
            self.summary_signal.emit(video_path, summary_text)
            
        self.progress_signal.emit(100)
        self.finished_signal.emit()

    def process_batch(self, images, timestamps, ai, db, video_path, model, processor, all_detected_indices):
        """Helper to run AI on a list of images at once"""
        try:
            inputs = processor(images=images, return_tensors="pt", padding=True).to(ai.device)
            
            with torch.no_grad():
                # Get Image Vectors
                img_features = model.get_image_features(**inputs)
                img_features /= img_features.norm(p=2, dim=-1, keepdim=True)
                
                # 1. Save Vectors to DB
                vectors_list = img_features.cpu().numpy().tolist()
                db.add_visual_embeddings(video_path, vectors_list, timestamps)
                
                # 2. Match Tags (Matrix Multiplication)
                # Shape: [Batch_Size, 512] @ [Tags_Count, 512].T -> [Batch_Size, Tags_Count]
                similarity = (100.0 * img_features @ ai.tag_embeddings.T).softmax(dim=-1)
                
                # Get Top 3 tags for EACH image in the batch
                top_vals, top_indices = similarity.topk(3, dim=-1)
                
                # Flatten indices and add to global counter
                all_detected_indices.extend(top_indices.cpu().numpy().flatten())
                
        except Exception as e:
            print(f"Batch Error: {e}")

    def stop(self):
        self.is_running = False

import os
import warnings
from PyQt6.QtCore import QThread, pyqtSignal
from core.ai_models import AIBackend
from core.database import Database

# Filter out Whisper's fp16 warning on CPU
warnings.filterwarnings("ignore")

class TranscriberWorker(QThread):
    log_signal = pyqtSignal(str)
    progress_signal = pyqtSignal(int)
    finished_signal = pyqtSignal()
    
    def __init__(self, file_paths, project_path):
        super().__init__()
        self.file_paths = file_paths
        self.project_path = project_path
        self.is_running = True

    def run(self):
        self.log_signal.emit("Loading Whisper Model...")
        ai = AIBackend()
        db = Database(self.project_path)
        
        try:
            model = ai.load_whisper()
        except Exception as e:
            self.log_signal.emit(f"Error loading model: {e}")
            self.finished_signal.emit()
            return

        total_files = len(self.file_paths)
        
        for idx, video_path in enumerate(self.file_paths):
            if not self.is_running: break
            
            file_name = os.path.basename(video_path)
            self.log_signal.emit(f"Transcribing: {file_name}...")
            
            try:
                # Transcribe with Word Timestamps
                segments, info = model.transcribe(video_path, beam_size=5, word_timestamps=False)
                
                transcript_data = []
                full_text_list = []
                
                for segment in segments:
                    # Save Rich Metadata (Start, End, Text)
                    transcript_data.append({
                        "start": round(segment.start, 2),
                        "end": round(segment.end, 2),
                        "text": segment.text.strip()
                    })
                    full_text_list.append(segment.text.strip())
                
                full_text = " ".join(full_text_list)
                
                if transcript_data:
                    # Save the STRUCTURED LIST to the database
                    # We pass the list 'transcript_data' instead of just a string
                    db.save_video_tags(video_path, [], transcript_data) 
                    self.log_signal.emit(f"✓ Transcribed {file_name}")
                else:
                    self.log_signal.emit(f"⚠ No speech found in {file_name}")
                    
            except Exception as e:
                self.log_signal.emit(f"Failed {file_name}: {str(e)}")

            percent = int(((idx + 1) / total_files) * 100)
            self.progress_signal.emit(percent)

        self.finished_signal.emit()

    def stop(self):
        self.is_running = False
